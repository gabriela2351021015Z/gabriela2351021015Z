async function enviarMensaje() { console.log("¡El botón funciona!"); // Esto debería salir en la consola al hacer clic const input = document.getElementById('userInput'); const chatBox = document.getElementById('chat-box'); const texto = input.value; if (!texto) { alert("El campo está vacío"); return; } chatBox.innerHTML += `

Tú: ${texto}

`; input.value = ''; try { console.log("Intentando conectar con:", 'https://whoops-name-walnut.ngrok-free.dev/webhook/tienda'); const response = await fetch('https://whoops-name-walnut.ngrok-free.dev/webhook/tienda', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ message: texto }) }); const data = await response.json(); console.log("Respuesta recibida:", data); // Esto nos dirá qué devuelve el servidor const respuestaBot = data.respuesta || data.output || "Sin respuesta"; chatBox.innerHTML += `

Bot: ${respuestaBot}

`; } catch (error) { console.error("Error capturado:", error); chatBox.innerHTML += `

Error en la conexión.

`; } }
